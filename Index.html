function generarIcono(tamano) {
    return new Promise(resolve => {
        const canvas = document.createElement('canvas');
        canvas.width = tamano;
        canvas.height = tamano;
        const ctx = canvas.getContext('2d');

        // Fondo verde con esquinas redondeadas
        const r = tamano * 0.2;
        ctx.beginPath();
        ctx.moveTo(r, 0);
        ctx.lineTo(tamano - r, 0);
        ctx.quadraticCurveTo(tamano, 0, tamano, r);
        ctx.lineTo(tamano, tamano - r);
        ctx.quadraticCurveTo(tamano, tamano, tamano - r, tamano);
        ctx.lineTo(r, tamano);
        ctx.quadraticCurveTo(0, tamano, 0, tamano - r);
        ctx.lineTo(0, r);
        ctx.quadraticCurveTo(0, 0, r, 0);
        ctx.closePath();

        const grad = ctx.createLinearGradient(0, 0, tamano, tamano);
        grad.addColorStop(0, '#2ecc71');
        grad.addColorStop(1, '#27ae60');
        ctx.fillStyle = grad;
        ctx.fill();

        // Icono de teléfono (texto emoji)
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        ctx.font = `${tamano * 0.5}px serif`;
        ctx.fillText('📞', tamano / 2, tamano / 2);

        canvas.toBlob(blob => resolve(blob), 'image/png');
    });
}

    './',

self.addEventListener('install', function (event) {
    event.waitUntil(
        caches.open(CACHE_NAME).then(function (cache) {
            return cache.addAll(ARCHIVOS);
        })
    );
    self.skipWaiting();
});

// Activación: borrar cachés antiguas
self.addEventListener('activate', function (event) {
    event.waitUntil(
        caches.keys().then(function (nombres) {
            return Promise.all(
                nombres
                    .filter(function (n) { return n !== CACHE_NAME; })
                    .map(function (n) { return caches.delete(n); })
            );
        })
    );
    self.clients.claim();
});

    );

function iniciarLlamada() {
    if (llamadaActiva) return;

    llamadaActiva = true;
    segundosLlamada = 0;

    // Actualizar interfaz
    document.getElementById('btn-llamar').classList.add('oculto');

    // Actualizar nombres en las tarjetas de llamada
    document.getElementById('nombre-hijo').textContent = contactos.hijo.nombre;

    document.getElementById('estado-hijo').textContent = 'Llamando...';
    document.getElementById('estado-hija').textContent = 'Llamando...';
    document.getElementById('estado-texto').textContent =
        `Llamando a ${contactos.hijo.nombre} y ${contactos.hija.nombre}...`;

    // Iniciar cronómetro
    timerInterval = setInterval(actualizarTiempo, 1000);

    // Vibrar el dispositivo si es posible
    if (navigator.vibrate) {
        navigator.vibrate([200, 100, 200]);
    }

    // Realizar las llamadas
    realizarLlamadas();

    mostrarNotificacion('📞 Llamando a tus hijos...');
}

function realizarLlamadas() {
    const telHijo = contactos.hijo.telefono.replace(/\s/g, '');
    const telHija = contactos.hija.telefono.replace(/\s/g, '');

    // --- Método: usar enlaces tel: ---
    // En móvil, "tel:" abre la app de teléfono directamente.
    // Para llamar a DOS personas simultáneamente en la web,
    // abrimos la primera llamada directamente y la segunda en nueva pestaña.

    // Primera llamada (hijo) - se abre directamente
    const linkHijo = document.createElement('a');
    linkHijo.href = `tel:${telHijo}`;
    linkHijo.click();

    // Segunda llamada (hija) - se abre con un pequeño retraso
    setTimeout(function () {
        const linkHija = document.createElement('a');
        linkHija.href = `tel:${telHija}`;
        linkHija.click();
    }, 2000);

    // Simular estado "conectado" tras unos segundos
    setTimeout(function () {
        if (llamadaActiva) {
            document.getElementById('estado-hijo').textContent = '✅ Conectado';
            document.getElementById('estado-hijo').style.color = '#2ecc71';
            document.getElementById('estado-hijo').style.animation = 'none';
        }
    }, 4000);

    setTimeout(function () {
        if (llamadaActiva) {
            document.getElementById('estado-hija').textContent = '✅ Conectada';
            document.getElementById('estado-hija').style.color = '#2ecc71';
            document.getElementById('estado-hija').style.animation = 'none';
            document.getElementById('estado-texto').textContent = '¡Conectado con tus hijos!';
        }
    }, 6000);
}

function colgarLlamada() {
    if (!llamadaActiva) return;

    llamadaActiva = false;

    // Detener cronómetro
    if (timerInterval) {
        clearInterval(timerInterval);
        timerInterval = null;
    }

    // Cerrar ventanas de llamada si se abrieron
    ventanasLlamada.forEach(function (ventana) {
        try {
            if (ventana && !ventana.closed) {
                ventana.close();
            }
        } catch (e) {
            // Ignorar errores de cierre
        }
    });
    ventanasLlamada = [];

    // Restaurar interfaz
    document.getElementById('btn-llamar').classList.remove('oculto');
    document.getElementById('btn-colgar').classList.add('oculto');
    document.getElementById('estado-llamada').classList.add('oculto');
    document.getElementById('contactos-llamada').classList.add('oculto');
    document.getElementById('info-contactos').classList.remove('oculto');

    // Restaurar estilos de estado
    const estadoHijo = document.getElementById('estado-hijo');
    const estadoHija = document.getElementById('estado-hija');
    estadoHijo.style.color = '';
    estadoHijo.style.animation = '';
    estadoHija.style.color = '';
    estadoHija.style.animation = '';

    // Resetear tiempo
    document.getElementById('tiempo-llamada').textContent = '00:00';
    segundosLlamada = 0;

    // Vibrar
    if (navigator.vibrate) {
        navigator.vibrate(300);
    }

    mostrarNotificacion(`📵 Llamada finalizada`);
}

function formatearTiempo(totalSegundos) {
    const minutos = Math.floor(totalSegundos / 60);
    const segundos = totalSegundos % 60;
    return `${String(minutos).padStart(2, '0')}:${String(segundos).padStart(2, '0')}`;
}

function guardarConfiguracion() {
    const nombreHijo = document.getElementById('input-nombre-hijo').value.trim();
    const telHijo = document.getElementById('input-tel-hijo').value.trim();
    const nombreHija = document.getElementById('input-nombre-hija').value.trim();
    const telHija = document.getElementById('input-tel-hija').value.trim();

    // Validaciones
    if (!nombreHijo || !telHijo || !nombreHija || !telHija) {
        mostrarNotificacion('⚠️ Complete todos los campos');
        return;
    }

    // Validar que los teléfonos tengan al menos 9 dígitos
    const soloDigitosHijo = telHijo.replace(/\D/g, '');
    const soloDigitosHija = telHija.replace(/\D/g, '');

    if (soloDigitosHijo.length < 9) {
        mostrarNotificacion('⚠️ El teléfono del hijo es muy corto');
        return;
    }

    if (soloDigitosHija.length < 9) {
        mostrarNotificacion('⚠️ El teléfono de la hija es muy corto');
        return;
    }

    // Guardar
    contactos.hijo.nombre = nombreHijo;
    contactos.hijo.telefono = telHijo;
    contactos.hija.nombre = nombreHija;
    contactos.hija.telefono = telHija;

    // Guardar en localStorage
    localStorage.setItem('contactos-hijos', JSON.stringify(contactos));

    // Actualizar vista
    actualizarVista();

    // Cerrar modal
    cerrarConfiguracion();

    mostrarNotificacion('✅ Contactos guardados correctamente');
}

        }
    }
}
